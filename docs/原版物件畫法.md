# 原版物件畫法紀錄

換成手繪蠟筆素材（assets/ui/*.png，原圖在 assets/source/）之前，這些物件都是程式直接畫的。下面是當時的 `_draw` 程式，取自 commit `5c65357`。
要換回原版：把對應檔案裡用 `draw_texture_rect` 畫圖的那段，換回下面這段即可（其餘動畫：外圈縮小、亮光、爆開等一直都是程式畫，沒有變）。

## 黃色音符

檔案：`scripts/level/rhythm_orb.gd`

```gdscript
func _draw() -> void:
	var lit := is_ready()
	var slot := _active_slot()
	## 大音符（跳得高）畫大、小音符（跳得矮）畫小；顏色一樣是黃色
	var base := COLOR
	var k := size_scale()
	var core := base if lit else Color(base.darkened(0.55), 0.55)
	if slot != -1 and slot == _used_slot:
		core = Color(0.5, 0.5, 0.55, 0.4)
	## 三連音用偏暖的外圈提示
	var ring_col := Color(1.0, 0.55, 0.4) if subdiv == 3 else base
	var until := _beats_to_next()
	if until < 1.0 and (_awake or not quiet):
		var t := 1.0 - until
		draw_arc(Vector2.ZERO, lerpf(RADIUS * 3.0, RADIUS, t) * k, 0.0, TAU, 40, Color(ring_col, 0.25 + 0.6 * t), 2.5, true)
	if lit:
		draw_circle(Vector2.ZERO, (RADIUS + 6.0) * k, Color(ring_col, 0.25))
	draw_circle(Vector2.ZERO, RADIUS * 0.62 * k, core)
	draw_arc(Vector2.ZERO, RADIUS * 0.62 * k, 0.0, TAU, 32, Color(1, 1, 1, 0.8 if lit else 0.3), 2.0, true)
	draw_line(Vector2(3, -9) * k, Vector2(3, 5) * k, Color(0.1, 0.08, 0.05, 0.8), 2.0 * k)
	draw_circle(Vector2(-1, 6) * k, 4.0 * k, Color(0.1, 0.08, 0.05, 0.8))
	if subdiv == 3:
		draw_string(ThemeDB.fallback_font, Vector2(-5, -RADIUS - 4), "3", HORIZONTAL_ALIGNMENT_LEFT, -1, 10, Color(ring_col, 0.85))
	if _burst > 0.0:
		draw_arc(Vector2.ZERO, RADIUS * k + (1.0 - _burst) * 30.0, 0.0, TAU, 40, Color(1, 1, 1, _burst), 3.0, true)


## 安靜模式（彈簧床飛出去之後要踩的音符／環用）：平常只是暗暗的不閃不響，
## 在彈簧床上彈的時候也不動；真的被彈出去、飛在附近的空中才「醒來」開始跟拍子亮、外圈縮小；落地就再睡。
var quiet: bool = false
var _awake: bool = false
```

```gdscript
func _draw() -> void:
```

## 起跳旗

檔案：`scripts/level/cue_flag.gd`

```gdscript
func _draw() -> void:
	var b := Conductor.song_beat
	var until := INF
	var since := INF
	if state == State.WATCH:
		until = go_beat() - GO_LEAD - b
	elif state == State.FREE:
		if _auto_waiting and _auto_go_at >= 0.0:
			until = _auto_go_at - GO_LEAD - b
			since = INF
		else:
			until = ceilf((b + GO_LEAD) / cycle_beats) * cycle_beats - GO_LEAD - b
			since = b + GO_LEAD - floorf((b + GO_LEAD) / cycle_beats) * cycle_beats
	draw_line(Vector2(0, 0), Vector2(0, -56), Color(0.85, 0.85, 0.9), 3.0)
	var fill := 0.25
	if until <= 2.0:
		fill = lerpf(0.5, 1.0, 1.0 - until / 2.0)
	elif state == State.WATCH:
		fill = 0.5
	var flash := clampf(1.0 - since * 3.0, 0.0, 1.0)
	var col := COLOR.lerp(Color.WHITE, flash)
	draw_colored_polygon(PackedVector2Array([Vector2(1, -56), Vector2(30, -47), Vector2(1, -38)]), Color(col, fill))
	_draw_countdown(until, since)
	if flash > 0.0:
		draw_arc(Vector2(0, -47), 20.0 + (1.0 - flash) * 24.0, 0.0, TAU, 32, Color(COLOR, flash), 3.0, true)
	if _prompt > 0.0:
		## 「↑ 示範」只在第一關提示，之後就不寫字了
		if GameState.level_index == 0:
			_draw_prompt()
	## 示範角色是關卡提示，不隨低畫質關掉（quality ghost 只影響其他裝飾）
	if state == State.WATCH and ghost.size() > 1:
		var t := b - start_beat
		if t >= 0.0 and t <= ghost_end() + 0.3:
			_draw_ghost(t)


## 倒數：響鈴前三拍跳出 3、2、1，響鈴那拍 GO!；下面四顆燈一拍亮一顆
```

## 鞦韆（繩子、座椅）

檔案：`scripts/level/swing_rope.gd`

```gdscript
func _draw() -> void:
	var b := Conductor.visual_beat()
	## 擺盪軌跡與「往前到頂」的星星
	var arc_pts := PackedVector2Array()
	for i in 25:
		var a := lerpf(-amplitude, amplitude, float(i) / 24.0)
		arc_pts.append(Vector2(sin(a), cos(a)) * length)
	draw_polyline(arc_pts, Color(1, 1, 1, 0.12), 2.0, true)
	var peak_local := Vector2(sin(dir * amplitude), cos(dir * amplitude)) * length
	var until := next_peak(b) - b
	var near := clampf(1.0 - until, 0.0, 1.0)
	var star_r := 7.0 + near * 5.0
	_draw_star(peak_local + Vector2(dir * 26.0, -26.0), star_r, Color(GLOW, 0.35 + 0.6 * near))
	## 錨點與繩子
	draw_circle(Vector2.ZERO, 7.0, Color(0.25, 0.2, 0.3))
	draw_circle(Vector2.ZERO, 3.5, ROPE)
	var seat_local := seat_at(b) - global_position
	var side := Vector2(cos(angle_at(b)), -sin(angle_at(b))) * SEAT_HALF
	draw_line(Vector2.ZERO, seat_local - side, ROPE, 2.5, true)
	draw_line(Vector2.ZERO, seat_local + side, ROPE, 2.5, true)
	## 座椅
	var plank := PackedVector2Array([seat_local - side, seat_local + side])
	draw_polyline(plank, SEAT.darkened(0.35), 9.0, true)
	draw_polyline(plank, SEAT, 6.0, true)
	if _rider == null:
		draw_circle(seat_local, GRAB_RADIUS * 0.55, Color(GLOW, 0.08 + 0.12 * near))
	else:
		## 搭乘中：一拍內從大縮到座椅的提示圈，縮到底就是放手的時機
		if until <= 1.0:
			var r := lerpf(64.0, 18.0, near * near)
			draw_arc(seat_local + Vector2(0, -20), r, 0.0, TAU, 40, Color(GLOW, 0.25 + 0.65 * near), 3.0, true)
	if _flash > 0.0:
		draw_circle(seat_local, 20.0 + (1.0 - _flash) * 50.0, Color(GLOW, _flash * 0.35))
```

## 尖刺（所有尖刺共用）

檔案：`scripts/level/spike.gd`

```gdscript
static func draw_spikes(canvas: CanvasItem, tile_size: float, height: float, color: Color) -> void:
	if height <= 0.01:
		return
	var w := tile_size / 3.0
	var tip_h := tile_size * 0.8 * height
	for i in 3:
		var x := i * w
		canvas.draw_colored_polygon(PackedVector2Array([
			Vector2(x + 1, tile_size), Vector2(x + w * 0.5, tile_size - tip_h), Vector2(x + w - 1, tile_size)]), color)
		canvas.draw_line(Vector2(x + w * 0.5, tile_size - tip_h), Vector2(x + w * 0.5, tile_size - tip_h * 0.6),
			Color(1, 1, 1, 0.6), 1.5)
```

```gdscript
func _draw() -> void:
	_visual_height = size.y
	var camera := get_viewport().get_camera_2d()
	if camera:
		var screen_bottom := camera.get_screen_center_position().y + get_viewport_rect().size.y / camera.zoom.y * 0.5
		_visual_height = maxf(_visual_height, screen_bottom + 128.0 - global_position.y)
	var cx := size.x * 0.5
	var b := Conductor.song_beat
	if is_open:
		_draw_open(cx, b)
	else:
		_draw_closed(cx, b)
	# 打開前一拍：兩側的光條往柱子收攏，收到柱子上那一刻就是拍子
	var until := BeatPattern.until_hit(notes, b)
	if until < 1.0:
		var t := 1.0 - until
		var d := lerpf(BRACKET_FROM, 13.0, t * t)
		var col := Color(OPEN_COLOR.lerp(Color.WHITE, 0.5 * t), 0.2 + 0.75 * t)
		for side: float in [-1.0, 1.0]:
			var x := cx + side * d
			draw_line(Vector2(x, 0), Vector2(x, _visual_height), col, 3.0)
			draw_line(Vector2(x, 1.5), Vector2(x - side * 9.0, 1.5), col, 3.0)
			draw_line(Vector2(x, _visual_height - 1.5), Vector2(x - side * 9.0, _visual_height - 1.5), col, 3.0)
	var cap := Color(0.85, 0.95, 1.0) if is_open else Color(0.55, 0.6, 0.8)
	draw_rect(Rect2(cx - 14, 0, 28, 4), cap)
```

## 彈簧床

檔案：`scripts/level/trampoline.gd`

```gdscript
func _draw() -> void:
	var lit := _rider != null
	## 腳架與外框
	for lx: float in [10.0, width * 0.5, width - 10.0]:
		draw_line(Vector2(lx - 6, 8), Vector2(lx - 10, tile), FRAME, 5.0)
		draw_line(Vector2(lx + 6, 8), Vector2(lx + 10, tile), FRAME, 5.0)
	draw_rect(Rect2(0, 6, width, 8), FRAME)
	draw_rect(Rect2(0, 6, width, 3), FRAME.lightened(0.25))
	## 床面：有人落下時往下凹
	var sag_x := width * 0.5
	if lit:
		sag_x = clampf(_rider.global_position.x - global_position.x, 12.0, width - 12.0)
	var sag := 9.0 * _sag
	var pts := PackedVector2Array()
	var n := 12
	for i in n + 1:
		var x := width * float(i) / float(n)
		var d := absf(x - sag_x) / maxf(width * 0.5, 1.0)
		pts.append(Vector2(x, 2.0 + sag * maxf(0.0, 1.0 - d * d * 3.0)))
	draw_polyline(pts, Color(COLOR, 0.95), 5.0, true)
	draw_polyline(pts, Color(1, 1, 1, 0.45), 1.5, true)
	## 彈簧
	for i in range(1, 6):
		var sx := width * float(i) / 6.0
		var top := pts[int(round(float(i) / 6.0 * n))].y + 2.0
		var zig := PackedVector2Array()
		for k in 5:
			zig.append(Vector2(sx + (3.0 if k % 2 == 0 else -3.0), lerpf(top, 6.0, float(k) / 4.0)))
		draw_polyline(zig, Color(0.8, 0.85, 0.95, 0.7), 1.5)
	## 段數燈：四顆
	var cx := width * 0.5
	## 連段模式只畫要按的次數（need 顆），一般模式畫四段
	var lights := need if need > 0 else TIERS.size()
	for i in lights:
		var on := lit and (i < _tier if need > 0 else i <= _tier) and not _waiting
		var pc := Vector2(cx + (i - (lights - 1) * 0.5) * 14.0, tile * 0.62)
		var off_col := Color(0.1, 0.08, 0.16, 0.85)
		if not lit and i == 0:
			off_col = off_col.lerp(COLOR, 0.25 + 0.5 * _sag)
		draw_circle(pc, 5.0, Color(COLOR, 0.95) if on else off_col)
		if on and _flash > 0.0:
			draw_circle(pc, 5.0 + _flash * 6.0, Color(COLOR, _flash * 0.5))
	## 下一次落床的提示圈：一拍內從大縮到床面；沒人時每拍都縮一次
	var land := ceilf(Conductor.song_beat)
	if lit:
		land = _next_beat if _waiting else _bounce_start + _bounce_len
	var until := land - Conductor.song_beat
	if until <= 1.0 and until >= 0.0:
		var k := 1.0 - until
		var r := lerpf(46.0, 14.0, k * k)
		var a := (0.25 + 0.6 * k) if lit else (0.12 + 0.3 * k)
		draw_arc(Vector2(sag_x, 2.0), r, PI, TAU, 24, Color(1, 1, 1, a), 2.5, true)
	if _flash > 0.0:
		draw_rect(Rect2(0, -2, width, 6), Color(1, 1, 1, _flash * 0.4))
```

```gdscript
## 小鼓：側面看的鼓身＋鼓皮；hit > 0 時鼓皮往下壓、外面擴一圈
func _draw_drum(at: Vector2, alpha: float, hit: float) -> void:
	var w := 170.0
	var h := 70.0
	var squash := 1.0 - 0.18 * hit
	var top := at + Vector2(0, -h * 0.5 * squash)
	var body := Rect2(at.x - w * 0.5, top.y, w, h * squash)
	var st := StyleBoxFlat.new()
	st.anti_aliasing = true
	st.set_corner_radius_all(14)
	st.bg_color = Color(UiKit.PINK, alpha)
	_view.draw_style_box(st, body)
	## 鼓身上的交叉繩
	for i in 5:
		var x0 := body.position.x + 10.0 + i * (w - 20.0) / 4.0
		_view.draw_line(Vector2(x0, body.position.y + 8), Vector2(x0 + 18.0, body.end.y - 6), Color(UiKit.NIGHT, 0.35 * alpha), 3.0, true)
	## 鼓皮（橢圓）
	var skin := PackedVector2Array()
	for i in 33:
		var a := TAU * i / 32.0
		skin.append(top + Vector2(cos(a) * w * 0.5, sin(a) * 18.0))
	_view.draw_colored_polygon(skin, Color(UiKit.TEXT, alpha))
	if hit > 0.0:
		var r := 100.0 + (1.0 - hit) * 60.0
		var col := Color(0.55, 1.0, 0.7) if absf(_hit_err) <= 0.04 else UiKit.YELLOW
		_view.draw_arc(top, r, PI, TAU, 40, Color(col, hit), 4.0, true)
```
